Живой пример · прозрачность

На тёмном фоне

Плашка прозрачная: видно только её саму, а фон просвечивает. Ниже та же капча лежит на тёмной панели — вокруг неё никакого белого прямоугольника.

Если бы капча рисовала белый фон, он бы закрыл тёмную панель. Здесь его нет.

Пример 1 · минимум

Три вставки

Окошко, кнопка и один обработчик. Это самый короткий рабочий вариант.

index.html
<!-- 1. окошко капчи -->
<iframe id="cap" src="https://doinwor.github.io/captcha/"
        width="500" height="180"
        style="border:0;background:transparent"
        sandbox="allow-scripts allow-same-origin allow-forms"></iframe>

<!-- 2. кнопка, которая ждёт прохождения -->
<button id="go" disabled>Продолжить</button>

<!-- 3. связать кнопку с капчей -->
<script>
  window.addEventListener('message', function (e) {
    if (e.origin !== 'https://doinwor.github.io') return;
    if (e.data.type === 'captcha-resize')  cap.style.height = e.data.height + 'px';
    if (e.data.type === 'captcha-success') go.disabled = false;
  });
</script>

Пример 3 · авто-высота

Окошко подстраивается само

Просим высоту сразу после загрузки и подгоняем окошко по сообщению captcha-resize.

JavaScript
var OK = 'https://doinwor.github.io';
var cap = document.getElementById('cap');

cap.addEventListener('load', function () {
  cap.contentWindow.postMessage({ type: 'captcha-height-request' }, OK);
});

window.addEventListener('message', function (e) {
  if (e.origin !== OK) return;
  var d = e.data || {};
  if (d.type === 'captcha-resize' && d.height) {
    cap.style.height = d.height + 'px';
  }
});
Пример 4 · отправка на сервер

Успех → запрос к бэкенду

После captcha-success отправляем данные формы на свой сервер.

JavaScript
window.addEventListener('message', function (e) {
  if (e.origin !== 'https://doinwor.github.io') return;
  if (!e.data || e.data.type !== 'captcha-success') return;

  // проверка пройдена — принимаем форму
  fetch('/api/signup', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email: email.value })
  }).then(function (r) { return r.json(); })
    .then(function () { location.href = '/thanks'; });
});
И на сервере

Клиентский сигнал легко подделать, поэтому на продакшене принимайте форму только после проверки на сервере. Больше — на странице Безопасность.

Пример 5 · модальное окно

Капча во всплывающем окне

Капчу можно показать в модалке и закрыть её после успеха.

JavaScript
function openCaptcha() {
  var modal = document.getElementById('modal');
  modal.classList.add('open');
  var f = modal.querySelector('iframe');
  if (!f.src) f.src = 'https://doinwor.github.io/captcha/'; // грузим один раз
}

window.addEventListener('message', function (e) {
  if (e.origin !== 'https://doinwor.github.io') return;
  if (e.data && e.data.type === 'captcha-success') {
    document.getElementById('modal').classList.remove('open');
    submitForm();   // форма, ради которой всё затевалось
  }
});
Пример 6 · несколько форм

Один обработчик на все капчи

Если капч на странице несколько, помечайте их атрибутами (data-cap) и обрабатывайте все одним слушателем — ровно так, как работает эта страница.

JavaScript
var OK = 'https://doinwor.github.io';
var caps = document.querySelectorAll('[data-cap]');

caps.forEach(function (f) {
  f.addEventListener('load', function () {
    try { f.contentWindow.postMessage({ type: 'captcha-height-request' }, OK); } catch (e) {}
  });
});

window.addEventListener('message', function (e) {
  if (e.origin !== OK) return;
  var d = e.data || {};

  if (d.type === 'captcha-resize' && d.height) {
    var h = Math.max(120, Math.min(520, d.height)) + 'px';
    caps.forEach(function (f) { f.style.height = h; });
  }
  if (d.type === 'captcha-success') {
    document.querySelectorAll('[data-unlock]').forEach(function (b) { b.disabled = false; });
  }
});
